    'labels'      => [ 'name' => 'Portfolio' ],
    'public'      => true,
    'show_in_rest' => true,
    'supports'    => [ 'title', 'editor', 'thumbnail' ],
    'has_archive' => true,
  ] );
} );

/**
 * ACF-powered flexible content builder.
 * Each section type maps to a template part.
 */
function shankar_render_blocks() {
  if( have_rows( 'content_blocks' ) ) :
    while( have_rows( 'content_blocks' ) ) : the_row();
      $layout = get_row_layout();
      get_template_part( "blocks/{$layout}" );
    endwhile;
  endif;
}

// WooCommerce: custom price logic with ACF
add_filter( 'woocommerce_product_get_price',
  function( $price, $product ) {
    $custom = get_field( 'custom_price', $product->get_id() );
    return $custom ?: $price;
  }, 10, 2
);
